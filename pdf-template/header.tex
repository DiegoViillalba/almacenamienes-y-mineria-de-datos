% pdf-template/header.tex
% Configuración de maquetación de libro académico con KOMA-Script (scrlayer-scrpage)
\usepackage{scrlayer-scrpage}

% Limpiar estilos de página previos
\clearpairofpagestyles

% Encabezados: Marca viva de capítulo o sección en el margen exterior
\ohead{\headmark}

% Pies de página: Número de página en el margen exterior (también en páginas de inicio de capítulo)
\ofoot[\pagemark]{\pagemark}

% Línea divisoria sutil bajo el encabezado
\KOMAoptions{headsepline=0.4pt}

% Estilos tipográficos sobrios para encabezados y pies
\setkomafont{pagehead}{\small\itshape}
\setkomafont{pagefoot}{\small}
\setkomafont{pagination}{}

% Formato de numeración en encabezados
\renewcommand*{\chaptermarkformat}{\thechapter.\ }
\renewcommand*{\sectionmarkformat}{\thesection.\ }

% Estilo de títulos de capítulos y secciones
\addtokomafont{chapter}{\sffamily\bfseries}
\addtokomafont{section}{\sffamily\bfseries}
\addtokomafont{subsection}{\sffamily\bfseries}

% Ajuste de tamaño máximo de imágenes para que no excedan el 70% del ancho del texto
\makeatletter
\AtBeginDocument{%
  \renewcommand*\pandocbounded[1]{%
    \sbox\pandoc@box{#1}%
    \Gscale@div\@tempa{\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
    \Gscale@div\@tempb{0.7\linewidth}{\wd\pandoc@box}%
    \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi% select the smaller of both
    \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
    \else\usebox{\pandoc@box}%
    \fi%
  }
}
\makeatother
